// HISTÓRICO DE VENDAS function carregarHistoricoVendas() { db.collection("historico_vendas").orderBy("numPedido", "desc").onSnapshot((snapshot) => { const tbody = document.getElementById('tabelaHistorico'); if (!tbody) return; tbody.innerHTML = ''; historicoVendasCache = []; let faturamento = 0; let totalPedidos = 0; snapshot.forEach((doc) => { const p = doc.data(); const id = doc.id; historicoVendasCache.push({ id, ...p }); faturamento += Number(p.total) || 0; totalPedidos++; tbody.innerHTML += ` #${p.numPedido} ${p.data} ${p.clienteNome} ${p.vendedor} ${p.formaPagamento} R$ ${Number(p.total).toFixed(2)} `; }); document.getElementById('fatTotal').innerText = `R$ ${faturamento.toFixed(2)}`; document.getElementById('totalQtdVendas').innerText = totalPedidos; }); }

Disponível: ${prod.estoque} un.`); return; } const itemExistente = carrinho.find(item => item.id === prodId); if (itemExistente) { if ((itemExistente.qtd + qtd) > prod.estoque) { alert("A quantidade total excede o estoque disponível."); return; } itemExistente.qtd += qtd; itemExistente.subtotal = itemExistente.qtd * itemExistente.preco; } else { carrinho.push({ id: prod.id, nome: prod.nome, preco: precoUn, qtd: qtd, subtotal: precoUn * qtd }); } atualizarTabelaCarrinho(); document.getElementById('carrinhoQtd').value = 1; } function removerDoCarrinho(index) { carrinho.splice(index, 1); atualizarTabelaCarrinho(); } function atualizarTabelaCarrinho() { const tbody = document.getElementById('tabelaCarrinho'); tbody.innerHTML = ''; let totalGeral = 0; carrinho.forEach((item, index) => { totalGeral += item.subtotal; tbody.innerHTML += ` ${item.nome} ${item.qtd} un.

async function salvarVendedor() { let nomeElem = document.getElementById('vendNome'); let senhaElem = document.getElementById('vendSenha'); if(!nomeElem || !senhaElem) return alert('Erro nos campos.'); let nome = nomeElem.value.trim(); let senha = senhaElem.value.trim(); if(!nome || !senha) return alert('Digite o nome e a senha do vendedor!'); let jaExiste = cacheVendedores.some(v => v.nome.toLowerCase() === nome.toLowerCase()); if(jaExiste) { return alert('Este vendedor já está cadastrado!'); } try { // Adiciona e TRAVA a execução até o Firebase confirmar que salvou await addDoc(collection(window.db, "vendedores"), { nome, senha }); nomeElem.value = ''; senhaElem.value = ''; // Recarrega todos os dados do banco antes de atualizar a tela await carregarTudoDoFirebase(); // Só depois desenha a lista nova carregarVendedores(); alert('Vendedor cadastrado com sucesso!'); } catch(e) { alert('Erro ao salvar: ' + e.message); } }
